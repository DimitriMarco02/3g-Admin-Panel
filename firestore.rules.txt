rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --- Helper Functions ---
    function isAuthenticated() {
      return request.auth != null;
    }

    function isUser(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function isAdmin() {
      return isAuthenticated() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true;
    }
    
    function isTeacher() {
      // Check for the existence of a mapping document. This is our source of truth for identifying teachers.
      return isAuthenticated() && exists(/databases/$(database)/documents/teacher_uids/$(request.auth.uid));
    }

    // Checks if the user is in the process of claiming a valid pending teacher account.
    // This is used when creating a document in the 'teacher_uids' collection.
    function isClaimingTeacherAccount() {
        // The client must provide the ID of the teacher document they are trying to claim.
        // This 'get' call verifies that the target document is pending and matches the user's email.
        let pendingTeacherDoc = get(/databases/$(database)/documents/teachers/$(request.resource.data.teacherDocId)).data;
        return pendingTeacherDoc.status == 'pending' && pendingTeacherDoc.email == request.auth.token.email;
    }

    // --- Publicly Readable Collections ---
    match /subjects/{doc=**}      { allow read: if true; allow write: if isAdmin(); }
    match /centers/{doc=**}       { allow read: if true; allow write: if isAdmin(); }
    match /slides/{doc=**}        { allow read: if true; allow write: if isAdmin(); }
    match /notices/{doc=**}       { allow read: if true; allow write: if isAdmin(); }
    match /diamondStudents/{doc=**} { allow read: if true; allow write: if isAdmin(); }
    match /settings/{doc=**}      { allow read: if true; allow write: if isAdmin(); }

    // --- New Teacher UID Mapping Collection ---
    // This collection securely links an authentication UID to a teacher document ID.
    match /teacher_uids/{userId} {
      allow read: if isUser(userId) || isAdmin();
      // Allow a user to create their OWN mapping doc IF they are claiming a valid pending account.
      allow create: if isUser(userId) && isClaimingTeacherAccount();
      // Only admins can modify or delete mappings after creation.
      allow update, delete: if isAdmin();
    }

    // --- Teachers Collection ---
    match /teachers/{teacherId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() 
                    || isOwnProfileUpdate()
                    || isClaimingUpdate()
                    || isAddingReview();
      
      // A teacher can update their own profile.
      function isOwnProfileUpdate() {
        return isTeacher() && request.auth.uid == resource.data.uid &&
               // A teacher cannot change their UID, email, or status via this self-service rule.
               request.resource.data.uid == resource.data.uid &&
               request.resource.data.email == resource.data.email &&
               request.resource.data.status == resource.data.status;
      }
      
      // A user signing up can update a pending teacher doc to claim it.
      function isClaimingUpdate() {
        let signingUpUser = request.auth;
        let oldData = resource.data;
        let newData = request.resource.data;
        
        return signingUpUser != null &&
          oldData.status == 'pending' &&
          oldData.email == signingUpUser.token.email &&
          newData.uid == signingUpUser.uid &&
          newData.status == 'active' &&
          // Ensure ONLY uid and status are changed in this specific operation.
          newData.diff(oldData).affectedKeys().hasOnly(['uid', 'status']);
      }
      
      // A student can add a review.
      function isAddingReview() {
        let isStudent = isAuthenticated() && !isTeacher() && !isAdmin();
        let onlyReviewsChanged = request.resource.data.diff(resource.data).affectedKeys().hasOnly(['reviews']);
        let reviewAdded = request.resource.data.reviews.size() == resource.data.reviews.size() + 1;
        return isStudent && onlyReviewsChanged && reviewAdded;
      }
    }

    // --- Users (Students/Admins) Collection ---
    match /users/{userId} {
      // MODIFIED: Added isTeacher() to allow teachers to read student profiles.
      allow read: if isAdmin() || isUser(userId) || isTeacher();
      allow create: if isUser(userId) && request.resource.data.isAdmin == false;
      allow update: if isAdmin() || 
                    (isUser(userId) && request.resource.data.isAdmin == resource.data.isAdmin
                                   && request.resource.data.studentId == resource.data.studentId);
      allow delete: if isAdmin();
    }

    // --- Bookings Collection ---
    match /bookings/{bookingId} {
      allow read: if isAdmin() || isUser(resource.data.userId) || isTeacherForBooking(resource.data.teacher.id);
      allow create: if isUser(request.resource.data.userId)
                    && (request.resource.data.status == 'Booked' || request.resource.data.status == 'Pending Admission');
      allow update: if isAdmin() || 
                    (isUser(resource.data.userId) && 
                     (request.resource.data.status == 'Canceled' || request.resource.data.status == 'Completed'));
      allow delete: if isAdmin();
      
      function isTeacherForBooking(teacherId) {
        let teacherDocId = get(/databases/$(database)/documents/teacher_uids/$(request.auth.uid)).data.teacherDocId;
        return isTeacher() && teacherId == teacherDocId;
      }
    }

    // --- Enrollments Collection ---
    match /enrollments/{enrollmentId} {
      allow read: if isAdmin() 
                  || isUser(resource.data.userId) 
                  || (isTeacher() && resource.data.teacherId == get(/databases/$(database)/documents/teacher_uids/$(request.auth.uid)).data.teacherDocId);
      allow write: if isAdmin();
    }
    
    // --- Payments Collection ---
    match /payments/{paymentId} {
      // Admin, the user who paid, or the teacher of the related enrollment can read.
      allow read: if isAdmin() 
                  || isUser(resource.data.userId) 
                  || isTeacherForPayment(resource.data.enrollmentId);

      // A student can submit a pending payment. An admin can create any payment.
      allow create: if (isUser(request.resource.data.userId) && request.resource.data.status == 'pending') || isAdmin();
      
      // Only admins can update (e.g., confirm a payment) or delete payments.
      allow update, delete: if isAdmin();

      // Helper to check if the requester is the teacher for the payment's enrollment.
      function isTeacherForPayment(enrollmentId) {
        let enrollment = get(/databases/$(database)/documents/enrollments/$(enrollmentId)).data;
        let teacherDocId = get(/databases/$(database)/documents/teacher_uids/$(request.auth.uid)).data.teacherDocId;
        return isTeacher() && enrollment.teacherId == teacherDocId;
      }
    }
    
    // --- Quizzes Collection ---
    match /quizzes/{quizId} {
      allow read: if true; // Public read for leaderboards and active quizzes
      allow write: if isAdmin();
    }

    // --- Quiz Submissions Collection ---
    match /quizSubmissions/{submissionId} {
      allow read: if true; // Public read for leaderboards
      allow create: if isUser(request.resource.data.userId); // Only users can create their own submissions
      allow update, delete: if isAdmin(); // Only admins can modify/delete submissions
    }

    // --- Chats Collection ---
    match /chats/{chatId} {
      // Allow a 'get' request if the user's ID is in the chatId.
      // This is secure and allows checking for existence before creating.
      allow get: if request.auth.uid in chatId.split('_');
      
      // 'list' (queries) and 'update' still require being a participant in the document's data.
      allow list, update: if request.auth.uid in resource.data.participants;
      
      // 'create' requires the user to be in the participants list of the new doc.
      allow create: if request.auth.uid in request.resource.data.participants;
      
      match /messages/{messageId} {
        allow read: if get(/databases/$(database)/documents/chats/$(chatId)).data.participants.hasAny([request.auth.uid]);
        allow create: if request.auth.uid == request.resource.data.senderId &&
                       get(/databases/$(database)/documents/chats/$(chatId)).data.participants.hasAny([request.auth.uid]);
      }
    }
  }
}
